    62→// Expert initialization function.
    63→int OnInit() {
    64→        // Perform core system check.
    65→        Sleep(100);
    66→        if (!RunTests()) {
    67→                Sleep(1000);
    68→                if (!RunTests()) {
    69→                        return(INIT_FAILED);
    70→                }
    71→        }
    72→
    73→        EventSetMillisecondTimer(1);  // Set Millisecond Timer to get client socket input
    74→
    75→        PrintFormat("[REP] Binding MT4 Server to Socket on Port %d.", REP_PORT);
    76→        PrintFormat("[PUSH] Binding MT4 Server to Socket on Port %d.", PUSH_PORT);
    77→
    78→        repSocket.bind(StringFormat("%s://%s:%d", ZEROMQ_PROTOCOL, HOSTNAME, REP_PORT));
    79→        pushSocket.bind(StringFormat("%s://%s:%d", ZEROMQ_PROTOCOL, HOSTNAME, PUSH_PORT));
    80→
    81→        // Maximum amount of time in milliseconds that the thread will try to send messages
    82→        // after its socket has been closed (the default value of -1 means to linger forever):
    83→
    84→        repSocket.setLinger(1000);  // 1000 milliseconds
    85→
    86→        /*      If we initiate socket.send() without having a corresponding socket draining the queue,
    87→                we'll eat up memory as the socket just keeps enqueueing messages.
    88→                So how many messages do we want ZeroMQ to buffer in RAM before blocking the socket?       */
    89→
    90→        repSocket.setSendHighWaterMark(10);  // 10 messages only.
    91→
    92→        return(INIT_SUCCEEDED);
    93→}

   104→// Expert timer function.
   105→void OnTimer() {
   106→        // Get client's response, but don't wait.
   107→        repSocket.recv(request, true);
   108→
   109→        // Send response to the client.
   110→        ZmqMsg reply = MessageHandler(request);

   114→        SendUpdateMessage(pushSocket);

   117→// Send Status Update data to the Client.
   118→void SendUpdateMessage(Socket &pSocket) {
   119→        int seconds = TimeSeconds(TimeCurrent());
   120→
   121→        // Update every second.
   122→        if (lastUpdateSeconds != seconds) {
   123→                lastUpdateSeconds = seconds;
   124→
   125→                if (subscribe_account) {
   126→                        string accountInfoString = GetAccountInfoString();
   127→                        if (lastUpdateAccount != accountInfoString) {
   128→                                lastUpdateAccount = accountInfoString;
   129→                                InformPullClient("STREAM", pSocket, "ACCOUNT", accountInfoString);
   130→                        }
   131→                }
   132→
   133→                if (subscribe_orders) {
   134→                        string ordersInfoString = GetAccountOrdersString();
   135→                        if (lastUpdateOrders != ordersInfoString) {
   136→                                lastUpdateOrders = ordersInfoString;
   137→                                InformPullClient("STREAM", pSocket, "ORDERS", ordersInfoString);
   138→                        }
   139→                }
   140→
   141→                if (ArraySize(subscribe_prices) > 0) {
   142→                        string changes[];
   143→                        for(int i = 0; i < ArraySize(subscribe_prices); i++) {
   144→                                string rateString = GetRatesString(subscribe_prices[i]);
   145→                                if (rateString != lastUpdateRates[i]) {
   146→                                        ArrayPush(changes, rateString);
   147→                                        lastUpdateRates[i] = rateString;
   148→                                }
   149→                        }
   150→                        string json = "[";
   151→                        for(int i = 0; i < ArraySize(changes); i++) {
   152→                                json += changes[i] + ",";
   153→                        }
   154→                        if (json != "[") {
   155→                                json = StringSubstr(json, 0, StringLen(json) - 1);
   156→                                json += "]";
   157→                                InformPullClient("STREAM", pSocket, "PRICES", StringFormat("%d|%s", RESPONSE_OK, json));
   158→                        }
   159→                }
   160→        }
   161→}

   163→// Handle request message.
   164→ZmqMsg MessageHandler(ZmqMsg &rRequest) {
   165→        // Output object.
   166→        ZmqMsg reply;
   167→
   168→        // Message components for later.
   169→        string components[];
   170→
   171→        if (request.size() > 0) {
   172→                // Get data from request.
   173→                ArrayResize(mdata, request.size());
   174→                request.getData(mdata);
   175→                string dataStr = CharArrayToString(mdata);
   176→
   177→                // Process data.
   178→                ParseZmqMessage(dataStr, components);
   179→
   180→                // Interpret data.
   181→                InterpretZmqMessage(&pushSocket, components);
   182→
   183→                // Construct response.
   184→                string id = components[1];
   185→                ZmqMsg ret(id);
   186→                reply = ret;
   187→        }
   188→
   189→        return(reply);
   190→}

   192→// Parse Zmq Message received from the client.
   193→// A message should be a string with values separated by pipe sign.
   194→// Example: 2|31|EURUSD
   195→void ParseZmqMessage(string& message, string& retArray[]) {
   196→        PrintFormat("Parsing: %s", message);
   197→        string sep = "|";
   198→        ushort u_sep = StringGetCharacter(sep, 0);
   199→        StringSplit(message, u_sep, retArray);
   200→}

   211→int RunTests() {

   213→
   214→        runningTests = true;

   216→        ///Checking for Permission to Perform Automated Trading for this program
   217→        if (!MQLInfoInteger(MQL_TRADE_ALLOWED) || !MQLInfoInteger(MQL_DLLS_ALLOWED))
   218→                testsNotPassed += 2;
   219→
   220→        // Checking if trading is allowed for any Expert Advisors/scripts for the current account
   221→        if (!AccountInfoInteger(ACCOUNT_TRADE_EXPERT))
   222→                testsNotPassed += 4;
   223→
   224→        // Checking if trading is allowed for the current account
   225→        if (!AccountInfoInteger(ACCOUNT_TRADE_ALLOWED))
   226→                testsNotPassed += 8;
   227→
   228→        // Should NOT ALLOW open price modification for market orders and pending orders out of freeze level range.
   229→        if (HasValidFreezeAndStopLevels("USDJPY", OP_BUY, Bid - 0.1, Bid, Bid - 0.3, Bid + 0.3) != -1 ||
   230→                HasValidFreezeAndStopLevels("USDJPY", OP_SELL, Ask + 0.1, Ask, Ask + 0.3, Ask - 0.3) != -1 ||
   231→                HasValidFreezeAndStopLevels("USDJPY", OP_BUYLIMIT, Bid - 0.1, Bid, Bid - 0.3, Bid + 0.3) != -2 ||
   232→                HasValidFreezeAndStopLevels("USDJPY", OP_SELLLIMIT, Ask + 0.1, Ask, Ask + 0.3, Ask - 0.3) != -2 ||
   233→                HasValidFreezeAndStopLevels("USDJPY", OP_BUYSTOP, Ask - 0.1, Ask + 0.1, Ask + 0.3, Ask + 0.3) != -2 ||
   234→                HasValidFreezeAndStopLevels("USDJPY", OP_SELLSTOP, Bid + 0.1, Bid - 0.1, Bid - 0.3, Bid - 0.5) != -2)
   235→                testsNotPassed += 16;
   236→
   237→        // Should NOT ALLOW stop loss modification for market orders out of freeze range.
   238→        if (HasValidFreezeAndStopLevels("USDJPY", OP_BUY, 0, Bid, Bid - 0.1, 0) != -3 ||
   239→                HasValidFreezeAndStopLevels("USDJPY", OP_SELL, Ask, Ask, Ask + 0.1, 0)!= -3)
   240→                testsNotPassed += 32;
   241→
   242→        // Should NOT ALLOW take profit modification for market orders out of freeze range.
   243→        if (HasValidFreezeAndStopLevels("USDJPY", OP_BUY, 0, Bid, 0, Bid + 0.1) != -4 ||
   244→                HasValidFreezeAndStopLevels("USDJPY", OP_SELL, Ask, Ask, 0, Ask - 0.1)!= -4)
   245→                testsNotPassed += 64;
   246→
   247→        // Should ALLOW open price modification for pending orders.
   248→        if (HasValidFreezeAndStopLevels("USDJPY", OP_BUYLIMIT, Bid - 0.5, Bid - 0.25, Bid - 0.6, Bid + 0.6) != 1 ||
   249→                HasValidFreezeAndStopLevels("USDJPY", OP_SELLLIMIT, Ask + 0.5, Ask + 0.25, Ask + 0.6, Ask - 0.6) != 1 ||
   250→                HasValidFreezeAndStopLevels("USDJPY", OP_BUYSTOP, Ask, Ask + 0.50, Ask + 0.25, Ask + 0.75) != 1 ||
   251→                HasValidFreezeAndStopLevels("USDJPY", OP_SELLSTOP, Bid, Bid - 0.50, Bid - 0.25, Bid - 0.75) != 1)
   252→                testsNotPassed += 128;
   253→
   254→        // Should calculate correct prices.
   255→        if (CalculateAndNormalizePrice("0", OP_BUY) != Ask ||
   256→                CalculateAndNormalizePrice(DoubleToStr(Ask), OP_BUYLIMIT) != Ask ||
   257→                CalculateAndNormalizePrice("-15", OP_SELLLIMIT) != Bid - 15 ||
   258→                CalculateAndNormalizePrice("+10", OP_BUYLIMIT) != Ask + 10 ||
   259→                CalculateAndNormalizePrice("%+10", OP_BUYLIMIT) != NormalizeDouble(Ask * 1.1, Digits) ||
   260→                CalculateAndNormalizePrice("%-15", OP_BUYLIMIT) != NormalizeDouble(Ask * 0.85, Digits))
   261→                testsNotPassed += 256;
   262→
   263→        runningTests = false;
   264→
   265→        if (testsNotPassed) {
   266→                PrintFormat("Tests didn't pass with %d!", testsNotPassed);
   267→                return false;
   268→        }
   269→
   270→        return true;
   271→}

   273→// Check order against exchange freeze and stop levels.
   274→//  1 - trade has valid open price and SL/TP.
   275→// -1 - open price modification is not allowed.
   276→// -2 - open price is out of freeze level range.
   277→// -3 - stop loss is out of freeze level range.
   278→// -4 - take profit is out of freeze level range.
   279→// -5 - open price is out of stop level range.
   280→// -6 - stop loss is out of stop level range.
   281→// -7 - take profit is out of stop level range.
   282→int HasValidFreezeAndStopLevels(string symbol, int cmd, double openPrice,
   283→        double price, double stoploss, double takeprofit) {
   284→
   285→        double FreezeLevel = runningTests ? 0.2 : MarketInfo(symbol, MODE_FREEZELEVEL);
   286→        double StopLevel = runningTests ? 0.2 : MarketInfo(symbol, MODE_STOPLEVEL);
   287→
   288→        if (openPrice != price) { // Check open price modification
   289→                if (openPrice && (cmd == OP_BUY || cmd == OP_SELL)) { // Check if market order
   290→                        return(-1);
   291→                } else if ( // Check freeze level
   292→                        (cmd == OP_BUYLIMIT && Ask - price <= FreezeLevel) ||
   293→                        (cmd == OP_SELLLIMIT && price - Bid <= FreezeLevel) ||
   294→                        (cmd == OP_BUYSTOP && price - Ask <= FreezeLevel) ||
   295→                        (cmd == OP_SELLSTOP && Bid - price <= FreezeLevel)
   296→                ) {
   297→                        return(-2);
   298→                } else if ( // Check stop level
   299→                        (cmd == OP_BUYLIMIT && Ask - price < StopLevel) ||
   300→                        (cmd == OP_SELLLIMIT && price - Bid < StopLevel) ||
   301→                        (cmd == OP_BUYSTOP && price - Ask < StopLevel) ||
   302→                        (cmd == OP_SELLSTOP && Bid - price < StopLevel)
   303→                ) {
   304→                        return(-5);
   305→                }
   306→        }
   307→
   308→        if (stoploss) { // Check stop loss
   309→                if ( // Check freeze level
   310→                        (cmd == OP_BUY && Bid - stoploss <= FreezeLevel) ||
   311→                        (cmd == OP_SELL && stoploss - Ask <= FreezeLevel)
   312→                ) {
   313→                        return(-3);
   314→                } else if ( // Check stop level
   315→                        (cmd == OP_BUY && Bid - stoploss < StopLevel) ||
   316→                        (cmd == OP_SELL && stoploss - Ask < StopLevel) ||
   317→                        (cmd == OP_BUYLIMIT && price - stoploss < StopLevel) ||
   318→                        (cmd == OP_SELLLIMIT && stoploss - price < StopLevel) ||
   319→                        (cmd == OP_BUYSTOP && price - stoploss < StopLevel) ||
   320→                        (cmd == OP_SELLSTOP && stoploss - price < StopLevel)
   321→                ) {
   322→                        return(-6);
   323→                }
   324→        }
   325→
   326→        if (takeprofit) { // Check take profit
   327→                if ( // Check freeze level
   328→                        (cmd == OP_BUY && takeprofit - Bid <= FreezeLevel) ||
   329→                        (cmd == OP_SELL && Ask - takeprofit <= FreezeLevel)
   330→                ) {
   331→                        return(-4);
   332→                } else if ( // Check stop level
   333→                        (cmd == OP_BUY && takeprofit - Bid < StopLevel) ||
   334→                        (cmd == OP_SELL && Ask - takeprofit < StopLevel) ||
   335→                        (cmd == OP_BUYLIMIT && takeprofit - price < StopLevel) ||
   336→                        (cmd == OP_SELLLIMIT && price - takeprofit < StopLevel) ||
   337→                        (cmd == OP_BUYSTOP && takeprofit - price < StopLevel) ||
   338→                        (cmd == OP_SELLSTOP && price - takeprofit < StopLevel)
   339→                ) {
   340→                        return(-7);
   341→                }
   342→        }
   343→
   344→        return 1;
   345→}

   347→// Apply optional modifier and normalize the price.
   348→// Possible modifiers are:
   349→//  - - reduce base or market price by given value
   350→//  + - increase base or market price by given value
   351→//  % - percentage multiplier of base or market price
   352→double CalculateAndNormalizePrice(string basePrice, int cmd) {
   353→        RefreshRates();
   354→
   355→        double price;
   356→        double marketPrice = cmd % 2 == 1 ? Bid : Ask;
   357→
   358→        if (cmd != -1 && (cmd < 2 || basePrice == "0")) { // Use market price
   359→                price = marketPrice;
   360→        } else if (cmd == -1 && basePrice == "0") { // Use empty value
   361→                price = 0;
   362→        } else {
   363→                string modifier = StringSubstr(basePrice, 0, 1);
   364→                if (modifier == "-") { // Undercut modifier
   365→                        price = marketPrice + StrToDouble(basePrice);
   366→                } else if (modifier == "+") { // Overcut modifier
   367→                        price = marketPrice + StrToDouble(StringSubstr(basePrice, 1));
   368→                } else if (modifier == "%") { // Percentage modifier
   369→                        price = marketPrice * (1 + (StrToDouble(StringSubstr(basePrice, 1))) / 100);
   370→                } else { // Apply no modifier
   371→                        price = StrToDouble(basePrice);
   372→                }
   373→
   374→                price = NormalizeDouble(price, Digits);
   375→        }
   376→
   377→        return price;
   378→}

   380→// Check the trade context status. Return codes:
   381→//  1 - trade context is free, trade allowed.
   382→//  0 - trade context was busy, but became free. Trade is allowed
   383→//      only after the market info has been refreshed.
   384→// -1 - trade context is busy, waiting interrupted by the user
   385→//      (expert was removed from the chart, terminal was shut down,
   386→//      the chart period and/or symbol was changed, etc.).
   387→// -2 - trade context is busy, waiting limit is reached (waitFor).
   388→int IsAllowedToTrade(int waitFor = 15) {
   389→        // check whether the trade context is free
   390→        if (!IsTradeAllowed()) {
   391→                int StartWaitingTime = (int)GetTickCount();
   392→
   393→                // infinite loop
   394→                while (true) {
   395→                        // if the expert was terminated by the user, stop operation
   396→                        if (IsStopped()) {
   397→                                Alert("The expert was terminated by the user!");
   398→                                return -1;
   399→                        }
   400→                        // if the waiting time exceeds the time specified in the
   401→                        // MaxWaiting_sec variable, stop operation, as well
   402→                        if ((int)GetTickCount() - StartWaitingTime > waitFor * 1000) {
   403→                                Alert(StringFormat("The waiting limit exceeded (%d seconds)!", waitFor));
   404→                                return -2;
   405→                        }
   406→                        // if the trade context has become free,
   407→                        if (IsTradeAllowed()) {
   408→                                return 0;
   409→                        }
   410→                        // if no loop breaking condition has been met, "wait" for 0.1
   411→                        // second and then restart checking
   412→                        Sleep(100);
   413→                }
   414→        } else {
   415→                return 1;
   416→        }
   417→}

   419→// Interpret Zmq Message validate it and perform an action.
   420→void InterpretZmqMessage(Socket& pSocket, string& compArray[]) {
   421→        // Pull data string.
   422→        string response = "";
   423→
   424→        // Message id and type of a message.
   425→        string apiKey = compArray[0];
   426→        string id = compArray[1];
   427→        int type = (int)compArray[2];
   428→
   429→        if (apiKey != APIKEY) {
   430→                response = StringFormat("%d|%s", RESPONSE_FAILED, getErrorReasonJson("Invalid API key"));
   431→        } else if (type == REQUEST_PING) {
   432→                response = StringFormat("%d|%d", RESPONSE_OK, TimeLocal());
   433→        } else if (type == REQUEST_TRADE_OPEN) {
   434→                if (ArraySize(compArray) != 13) {
   435→                        response = StringFormat("%d|%s", RESPONSE_FAILED, getErrorJson(4050));
   436→                } else {
   437→                        string symbol = compArray[3];
   438→                        int operation = (int)compArray[4];
   439→                        double volume = (double)compArray[5];
   440→                        string basePrice = compArray[6];
   441→                        int slippage = (int)compArray[7];
   442→                        string baseStoploss = compArray[8];
   443→                        string baseTakeprofit = compArray[9];
   444→                        string comment = compArray[10];
   445→                        int magicnumber = (int)compArray[11];
   446→                        int unit = (int)compArray[12];
   447→
   448→                        response = OpenOrder(symbol, operation, volume, basePrice, slippage, baseStoploss, baseTakeprofit, comment, magicnumber, unit);
   449→                }
   450→        } else if (type == REQUEST_TRADE_MODIFY) {
   451→                if (ArraySize(compArray) != 7) {
   452→                        response = StringFormat("%d|%s", RESPONSE_FAILED, getErrorJson(4050));
   453→                } else {
   454→                        int ticket = (int)compArray[3];
   455→                        string basePrice = compArray[4];
   456→                        string baseStoploss = compArray[5];
   457→                        string baseTakeprofit = compArray[6];
   458→
   459→                        response = ModifyOrder(ticket, basePrice, baseStoploss, baseTakeprofit);
   460→                }
   461→        } else if (type == REQUEST_TRADE_DELETE) {
   462→                if (ArraySize(compArray) != 4) {
   463→                        response = StringFormat("%d|%s", RESPONSE_FAILED, getErrorJson(4050));
   464→                } else {
   465→                        int ticket = (int)compArray[3];
   466→                        response = DeletePendingOrder(ticket);
   467→                }
   468→        } else if (type == REQUEST_RATES) {
   469→                if (ArraySize(compArray) != 4) {
   470→                        response = StringFormat("%d|%s", RESPONSE_FAILED, getErrorJson(4050));
   471→                } else {
   472→                        string symbol = compArray[3];
   473→                        response = GetRatesString(symbol);
   474→                }
   475→        } else if (type == REQUEST_ACCOUNT) {
   476→                response = GetAccountInfoString();
   477→        } else if (type == REQUEST_ORDERS) {
   478→                response = GetAccountOrdersString();
   479→        } else if (type == REQUEST_DELETE_ALL_PENDING_ORDERS) {
   480→                if (ArraySize(compArray) != 4) {
   481→                        response = StringFormat("%d|%s", RESPONSE_FAILED, getErrorJson(4050));
   482→                } else {
   483→                        string symbol = compArray[3];
   484→                        response = DeleteAllPendingOrders(symbol);
   485→                }
   486→        } else if (type == REQUEST_CLOSE_MARKET_ORDER) {
   487→                if (ArraySize(compArray) != 4) {
   488→                        response = StringFormat("%d|%s", RESPONSE_FAILED, getErrorJson(4050));
   489→                } else {
   490→                        int ticket = (int)compArray[3];
   491→                        response = CloseMarketOrder(ticket);
   492→                }
   493→        } else if (type == REQUEST_CLOSE_ALL_MARKET_ORDERS) {
   494→                if (ArraySize(compArray) != 4) {
   495→                        response = StringFormat("%d|%s", RESPONSE_FAILED, getErrorJson(4050));
   496→                } else {
   497→                        string symbol = compArray[3];
   498→                        response = CloseAllMarketOrders(symbol);
   499→                }
   500→        } else if (type == REQUEST_SUBSCRIBE_ACCOUNT) {
   501→                subscribe_account = true;
   502→                response = StringFormat("%d|%s", RESPONSE_OK, "{\"status\":true}");
   503→        } else if (type == REQUEST_SUBSCRIBE_ORDERS) {
   504→                subscribe_orders = true;
   505→                response = StringFormat("%d|%s", RESPONSE_OK, "{\"status\":true}");
   506→        } else if (type == REQUEST_SUBSCRIBE_PRICES) {
   507→                string symbols[];
   508→                string sep = ",";
   509→                ushort u_sep = StringGetCharacter(sep, 0);
   510→        StringSplit(compArray[3], u_sep, symbols);
   511→                for (int i = 0; i < ArraySize(symbols); i++) {
   512→                        bool found = false;
   513→                        for (int i2 = 0; i2 < ArraySize(subscribe_prices); i2++) {
   514→                                if (symbols[i] == subscribe_prices[i2]) {
   515→                                        found = true;

   519→                        if (found == false) {
   520→                                ArrayPush(subscribe_prices, symbols[i]);

   525→        } else if (type == REQUEST_UNSUBSCRIBE_ACCOUNT) {
   526→                subscribe_account = false;

   529→                subscribe_orders = false;

   533→                string sep = ",";
   534→                ushort u_sep = StringGetCharacter(sep, 0);
   535→        StringSplit(compArray[3], u_sep, symbols);

   539→                                        RemoveIndexFromArray(subscribe_prices,i2);

   541→                                        break;
   542→                                }
   543→                        }
   544→                }
   545→                string status = "true";
   546→                for (int i = 0; i < ArraySize(symbols); i++) {
   547→                for (int i2 = 0; i2 < ArraySize(subscribe_prices); i2++) {
   548→                        if (symbols[i] == subscribe_prices[i2]) {
   549→                                status = "false";
   550→                                break;
   551→                        }
   552→                }
   553→        }
   554→                response = StringFormat("%d|%s", RESPONSE_OK, "{\"status\":" + status + "}");
   555→        } else if (type == REQUEST_CHART) {
   556→                bool opened = ChartOpen(compArray[3], 1) != 0;
   557→                if (opened) {
   558→                        Sleep(100);
   559→                        MqlRates rates[];
   560→                        ArrayCopyRates(rates, compArray[3], 1);
   561→                        closeAllChart();
   562→                        string json = ratesToJson(rates);
   563→                        response = StringFormat("%d|%s", RESPONSE_OK, json);
   564→                } else {
   565→                        response = StringFormat("%d|%s", RESPONSE_FAILED, "{\"error\":\"ChartOpen failed\"}");
   566→                }
   567→        } else {
   568→                response = StringFormat("%d|%s", RESPONSE_FAILED, "{\"error\":\"invalid type\"}");
   569→        }
   570→
   571→        // Send a response.
   572→        InformPullClient("RESPONSE", pSocket, id, response);

   575→// Open an order.
   576→// Input:       SYMBOL|OPERATION|VOLUME|BASE_PRICE|SLIPPAGE|
   577→//                               BASE_STOPLOSS|BASE_TAKEPROFIT|COMMENT|MAGIC_NUMBER|
   578→//                               UNIT
   579→// Example: USDJPY|2|1|108.848|0|0|0|some text|123|0
   580→// Output:  TICKET
   581→// Example: 140602286
   582→string OpenOrder(string symbol, int cmd, double volume, string basePrice, int slippage, string baseStoploss,
   583→        string baseTakeprofit, string comment, int magicnumber, int unit) {
   584→        while(true) {
   585→                double price = CalculateAndNormalizePrice(basePrice, cmd);
   586→                double stoploss = CalculateAndNormalizePrice(baseStoploss, -1);
   587→                double takeprofit = CalculateAndNormalizePrice(baseTakeprofit, -1);
   588→
   589→                if (unit == UNIT_CURRENCY) {
   590→                        volume /= price;
   591→                }
   592→
   593→                if (HasValidFreezeAndStopLevels(symbol, cmd, 0, price, stoploss, takeprofit) < 0) {
   594→                        return StringFormat("%d|%s", RESPONSE_FAILED, getErrorJson(130)); // Invalid stops
   595→                } else if (volume < MarketInfo(symbol, MODE_MINLOT) || volume > MarketInfo(symbol, MODE_MAXLOT)) {
   596→                        return StringFormat("%d|%s", RESPONSE_FAILED, getErrorJson(131)); // Volume below min or above max
   597→                } else if (cmd < 2 && (AccountFreeMarginCheck(symbol, cmd, volume) <= 0 || GetLastError() == 134)) {
   598→                        return StringFormat("%d|%s", RESPONSE_FAILED, getErrorJson(134)); // Free margin is insufficient
   599→                } else if (IsAllowedToTrade() < 0) {
   600→                        return StringFormat("%d|%s", RESPONSE_FAILED, getErrorJson(146)); // Trader context is busy
   601→                }
   602→
   603→                int ticket = OrderSend(symbol, cmd, volume, price, slippage, stoploss, takeprofit, comment, magicnumber, 0, clrGreen);
   604→
   605→                if (ticket > 0) {
   606→                        string json = "{\"ticket\":\"" + IntegerToString(ticket) + "\"}";
   607→                        return StringFormat("%d|%s", RESPONSE_OK, json);
   608→                } else {
   609→                        int error = GetLastError();
   610→
   611→                        switch(error) {                                                          // Overcomable errors
   612→                                case 4:                                                                         // Trade server is busy. Retrying...
   613→                                        Sleep(2000);                                                     // Try again
   614→                                        continue;                                                                // At the next iteration
   615→                                case 135:                                                                        // The price has changed. Retrying...
   616→                                        RefreshRates();                                          // Update data
   617→                                        continue;                                                                // At the next iteration
   618→                                case 136:                                                                        // No prices. Waiting for a new tick...
   619→                                        while(RefreshRates() == false)   // Up to a new tick
   620→                                                Sleep(1);                                                        // Cycle delay
   621→                                        continue;                                                                // At the next iteration
   622→                                case 137:                                                                        // Broker is busy. Retrying...
   623→                                        Sleep(2000);                                                     // Try again
   624→                                        continue;                                                                // At the next iteration
   625→                                case 146:                                                                        // Trading subsystem is busy. Retrying...
   626→                                        Sleep(500);                                                       // Try again
   627→                                        RefreshRates();                                          // Update data
   628→                                        continue;                                                                // At the next iteration
   629→                                default:
   630→                                        Alert(StringFormat("OpenOrder error: %d", error));
   631→                                        return(StringFormat("%d|%s", RESPONSE_FAILED, getErrorJson(error)));
   632→                        }
   633→                }
   634→        }
   635→}
